{{/* Site OG preview lookup. The generated preview files carry the site's default language suffix (ogPreview.de.png, see scripts/preview.sh + theme's preview.py), so the language is always part of the file name — unlike the theme's prerendered.html, which only adds it for secondary sites. */}} {{- $previewImg := "" -}} {{- if isset site.Params "preview" -}} {{- $previewImgPrefix := printf "%s.%s%s" site.Params.preview.outputPrefix site.Language.Lang site.Params.preview.outputSuffix -}} {{- $previewImg = (.context.Resources.ByType "image").GetMatch $previewImgPrefix -}} {{- end -}} {{- return $previewImg -}}
